DECLARE @Search NVARCHAR(4000) = N'SZUKANA_WARTOŚĆ';
DECLARE @SQL NVARCHAR(MAX) = N'';

SELECT @SQL += '
SELECT
    ''' + s.name + '.' + t.name + ''' AS TableName,
    ''' + c.name + ''' AS ColumnName,
    CAST([' + c.name + '] AS NVARCHAR(MAX)) AS FoundValue
FROM [' + s.name + '].[' + t.name + ']
WHERE CAST([' + c.name + '] AS NVARCHAR(MAX)) LIKE N''%' + REPLACE(@Search,'''','''''') + '%''
UNION ALL
'
FROM sys.tables t
JOIN sys.schemas s ON t.schema_id = s.schema_id
JOIN sys.columns c ON t.object_id = c.object_id
JOIN sys.types ty ON c.user_type_id = ty.user_type_id
WHERE ty.name IN ('varchar','nvarchar','char','nchar','text','ntext');

-- Usuń ostatnie UNION ALL
SET @SQL = LEFT(@SQL, LEN(@SQL) - LEN('UNION ALL' + CHAR(13) + CHAR(10)));

EXEC sp_executesql @SQL;
